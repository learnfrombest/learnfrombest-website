import React, { useState, useEffect } from 'react';

const INITIAL_PROJECTS = [
  { id: 'PRJ-01', name: 'Intelligent Document Processing', desc: 'OCR & Bedrock pipeline for multi-format invoices.', status: 'In Progress' },
  { id: 'PRJ-02', name: 'Enterprise RAG Knowledge Assistant', desc: 'Bedrock Knowledge Base + OpenSearch retrieval.', status: 'In Progress' },
  { id: 'PRJ-03', name: 'Fraud & Anomaly Detection', desc: 'Real-time transaction risk scoring with Lambda.', status: 'Planning' },
];

const INITIAL_TASKS = [
  {
    id: 'LFB-101',
    projectId: 'PRJ-01',
    title: 'Extract tabular line items via AWS Textract',
    description: 'Parse multi-page PDF statements into JSON schema.',
    status: 'In Progress',
    priority: 'High',
    assignee: 'Lead Architect',
    proof: null,
    comments: [
      { id: 'c1', author: 'Lead Architect', text: 'Textract expense analysis integration started.', time: '10:14 AM' }
    ]
  },
  {
    id: 'LFB-102',
    projectId: 'PRJ-01',
    title: 'Validate invoice tax totals against line sum',
    description: 'Ensure calculated arithmetic matches billed totals.',
    status: 'Done',
    priority: 'High',
    assignee: 'Lead Architect',
    proof: { name: 'test_pass_validation.png', data: null },
    comments: [
      { id: 'c2', author: 'Lead Architect', text: 'Task is complete. All 15 sample invoices matched with 0 mismatch.', isSignOff: true, time: '11:30 AM' }
    ]
  },
  {
    id: 'LFB-103',
    projectId: 'PRJ-02',
    title: 'Configure OpenSearch Serverless Vector Index',
    description: 'Set dimension 1536 for Titan Embeddings.',
    status: 'To Do',
    priority: 'Medium',
    assignee: 'Lead Architect',
    proof: null,
    comments: []
  }
];

export default function ProjectStudio({ onSignOut, currentUserEmail }) {
  const [projects, setProjects] = useState(() => {
    const saved = localStorage.getItem('lfb_studio_projects');
    return saved ? JSON.parse(saved) : INITIAL_PROJECTS;
  });

  const [tasks, setTasks] = useState(() => {
    const saved = localStorage.getItem('lfb_studio_tasks');
    return saved ? JSON.parse(saved) : INITIAL_TASKS;
  });

  const [selectedProjectId, setSelectedProjectId] = useState('PRJ-01');
  const [activeTask, setActiveTask] = useState(null);
  const [commentInput, setCommentInput] = useState('');
  const [proofFile, setProofFile] = useState(null);
  const [proofFileName, setProofFileName] = useState('');
  const [showNewTaskModal, setShowNewTaskModal] = useState(false);
  const [newTaskTitle, setNewTaskTitle] = useState('');
  const [newTaskDesc, setNewTaskDesc] = useState('');
  const [newTaskPriority, setNewTaskPriority] = useState('High');

  useEffect(() => {
    localStorage.setItem('lfb_studio_projects', JSON.stringify(projects));
  }, [projects]);

  useEffect(() => {
    localStorage.setItem('lfb_studio_tasks', JSON.stringify(tasks));
  }, [tasks]);

  const activeProject = projects.find(p => p.id === selectedProjectId) || projects[0];
  const projectTasks = tasks.filter(t => t.projectId === selectedProjectId);

  const handleFileUpload = (e) => {
    const file = e.target.files[0];
    if (!file) return;
    setProofFileName(file.name);
    const reader = new FileReader();
    reader.onload = (event) => setProofFile(event.target.result);
    reader.readAsDataURL(file);
  };

  const submitComment = (isQuickComplete = false) => {
    if (!activeTask) return;
    const author = currentUserEmail || 'Workspace User';
    const text = isQuickComplete
      ? 'Task is complete and verified with proof of execution.'
      : commentInput.trim();

    if (!text && !proofFile) return;

    const newComment = {
      id: 'c_' + Date.now(),
      author,
      text,
      isSignOff: isQuickComplete,
      time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
      proofAttachment: proofFile ? { name: proofFileName, data: proofFile } : null
    };

    const updatedTask = {
      ...activeTask,
      status: isQuickComplete ? 'Done' : activeTask.status,
      proof: proofFile ? { name: proofFileName, data: proofFile } : activeTask.proof,
      comments: [...activeTask.comments, newComment]
    };

    const updatedTasks = tasks.map(t => t.id === activeTask.id ? updatedTask : t);
    setTasks(updatedTasks);
    setActiveTask(updatedTask);
    setCommentInput('');
    setProofFile(null);
    setProofFileName('');
  };

  const handleCreateTask = () => {
    if (!newTaskTitle.trim()) return;
    const newTask = {
      id: `LFB-${100 + tasks.length + 1}`,
      projectId: selectedProjectId,
      title: newTaskTitle.trim(),
      description: newTaskDesc.trim(),
      status: 'To Do',
      priority: newTaskPriority,
      assignee: currentUserEmail || 'Workspace User',
      proof: null,
      comments: []
    };
    setTasks([newTask, ...tasks]);
    setNewTaskTitle('');
    setNewTaskDesc('');
    setShowNewTaskModal(false);
  };

  const columns = ['To Do', 'In Progress', 'In Review', 'Done'];

  return (
    <div style={{ minHeight: '100vh', background: '#0b1120', color: '#f1f5f9', fontFamily: 'Inter, system-ui, sans-serif' }}>
      {/* Top Header */}
      <header style={{ height: '64px', borderBottom: '1px solid #1e293b', background: '#0f172a', display: 'flex', alignItems: 'center', justifyContent: 'space-between', padding: '0 24px' }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: '16px' }}>
          <span style={{ background: '#3b82f6', color: '#fff', fontWeight: 800, padding: '6px 12px', borderRadius: '8px', fontSize: '13px' }}>LFB</span>
          <h1 style={{ fontSize: '16px', margin: 0, fontWeight: 700 }}>Project & Task Studio</h1>
          <select
            value={selectedProjectId}
            onChange={(e) => setSelectedProjectId(e.target.value)}
            style={{ background: '#1e293b', color: '#f8fafc', border: '1px solid #334155', borderRadius: '6px', padding: '6px 12px', fontSize: '13px' }}
          >
            {projects.map(p => <option key={p.id} value={p.id}>{p.name} ({p.id})</option>)}
          </select>
        </div>
        <div style={{ display: 'flex', alignItems: 'center', gap: '12px' }}>
          <button
            onClick={() => setShowNewTaskModal(true)}
            style={{ background: '#2563eb', color: '#fff', border: 0, padding: '7px 14px', borderRadius: '6px', fontSize: '13px', fontWeight: 600, cursor: 'pointer' }}
          >
            ＋ Create Task
          </button>
          {onSignOut && (
            <button
              onClick={onSignOut}
              style={{ background: 'transparent', color: '#94a3b8', border: '1px solid #334155', padding: '6px 12px', borderRadius: '6px', fontSize: '12px', cursor: 'pointer' }}
            >
              Sign Out
            </button>
          )}
        </div>
      </header>

      {/* Main Kanban Board */}
      <main style={{ padding: '24px', maxWidth: '1600px', margin: '0 auto' }}>
        <div style={{ display: 'grid', gridTemplateColumns: 'repeat(4, 1fr)', gap: '18px' }}>
          {columns.map(col => {
            const colTasks = projectTasks.filter(t => t.status === col);
            return (
              <div key={col} style={{ background: '#0f172a', border: '1px solid #1e293b', borderRadius: '10px', padding: '16px', minHeight: '650px' }}>
                <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', marginBottom: '14px' }}>
                  <h3 style={{ fontSize: '13px', fontWeight: 700, margin: 0, textTransform: 'uppercase', letterSpacing: '0.05em', color: '#94a3b8' }}>
                    {col}
                  </h3>
                  <span style={{ background: '#1e293b', color: '#cbd5e1', padding: '2px 8px', borderRadius: '12px', fontSize: '11px', fontWeight: 600 }}>
                    {colTasks.length}
                  </span>
                </div>

                <div style={{ display: 'flex', flexDirection: 'column', gap: '10px' }}>
                  {colTasks.map(task => (
                    <div
                      key={task.id}
                      onClick={() => setActiveTask(task)}
                      style={{ background: '#1e293b', border: '1px solid #334155', borderRadius: '8px', padding: '14px', cursor: 'pointer' }}
                    >
                      <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '11px', color: '#64748b', marginBottom: '6px' }}>
                        <span>{task.id}</span>
                        <span style={{ color: task.priority === 'High' ? '#f87171' : '#fbbf24', fontWeight: 600 }}>{task.priority}</span>
                      </div>
                      <h4 style={{ margin: '0 0 10px', fontSize: '13px', lineHeight: 1.4, color: '#f8fafc' }}>{task.title}</h4>
                      
                      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', fontSize: '11px', color: '#94a3b8' }}>
                        <span>{task.comments.length} comments</span>
                        {task.proof && (
                          <span style={{ background: '#064e3b', color: '#34d399', padding: '2px 6px', borderRadius: '4px', fontWeight: 600 }}>
                            ✓ Proof attached
                          </span>
                        )}
                      </div>
                    </div>
                  ))}
                  {colTasks.length === 0 && (
                    <div style={{ textAlign: 'center', padding: '40px 10px', color: '#475569', fontSize: '12px' }}>
                      No tasks in this lane
                    </div>
                  )}
                </div>
              </div>
            );
          })}
        </div>
      </main>

      {/* Task Detail & Proof Modal */}
      {activeTask && (
        <div style={{ position: 'fixed', inset: 0, background: 'rgba(0,0,0,0.7)', display: 'flex', justifyContent: 'center', alignItems: 'center', zIndex: 50, padding: '20px' }}>
          <div style={{ background: '#0f172a', border: '1px solid #334155', borderRadius: '12px', width: '100%', maxWidth: '720px', maxHeight: '90vh', overflowY: 'auto', padding: '24px', display: 'flex', flexDirection: 'column', gap: '18px' }}>
            <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
              <div>
                <span style={{ fontSize: '12px', color: '#64748b', fontWeight: 600 }}>{activeTask.id} · {activeProject.name}</span>
                <h2 style={{ margin: '4px 0 0', fontSize: '18px', color: '#f8fafc' }}>{activeTask.title}</h2>
              </div>
              <button
                onClick={() => { setActiveTask(null); setProofFile(null); setProofFileName(''); }}
                style={{ background: '#1e293b', border: '1px solid #334155', color: '#94a3b8', borderRadius: '6px', width: '32px', height: '32px', cursor: 'pointer', fontSize: '16px' }}
              >
                ✕
              </button>
            </div>

            <div style={{ display: 'grid', gridTemplateColumns: 'repeat(3, 1fr)', gap: '12px', background: '#1e293b', padding: '12px', borderRadius: '8px', fontSize: '12px' }}>
              <div>
                <span style={{ color: '#64748b', display: 'block', marginBottom: '2px' }}>Status</span>
                <select
                  value={activeTask.status}
                  onChange={(e) => {
                    const updated = { ...activeTask, status: e.target.value };
                    setActiveTask(updated);
                    setTasks(tasks.map(t => t.id === activeTask.id ? updated : t));
                  }}
                  style={{ background: '#0f172a', color: '#f8fafc', border: '1px solid #334155', borderRadius: '4px', padding: '4px 8px', width: '100%' }}
                >
                  {columns.map(c => <option key={c} value={c}>{c}</option>)}
                </select>
              </div>
              <div>
                <span style={{ color: '#64748b', display: 'block', marginBottom: '2px' }}>Priority</span>
                <span style={{ fontWeight: 600, color: activeTask.priority === 'High' ? '#f87171' : '#fbbf24' }}>{activeTask.priority}</span>
              </div>
              <div>
                <span style={{ color: '#64748b', display: 'block', marginBottom: '2px' }}>Assignee</span>
                <span style={{ color: '#cbd5e1' }}>{activeTask.assignee}</span>
              </div>
            </div>

            <div>
              <h4 style={{ margin: '0 0 6px', fontSize: '12px', textTransform: 'uppercase', letterSpacing: '0.05em', color: '#94a3b8' }}>Description</h4>
              <p style={{ margin: 0, fontSize: '13px', color: '#cbd5e1', lineHeight: 1.5, background: '#1e293b', padding: '12px', borderRadius: '8px' }}>
                {activeTask.description || 'No description provided.'}
              </p>
            </div>

            {activeTask.proof && (
              <div style={{ background: '#064e3b', border: '1px solid #059669', borderRadius: '8px', padding: '12px' }}>
                <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between' }}>
                  <span style={{ fontSize: '12px', color: '#a7f3d0', fontWeight: 600 }}>✓ Attached Proof: {activeTask.proof.name}</span>
                  {activeTask.proof.data && (
                    <a
                      href={activeTask.proof.data}
                      download={activeTask.proof.name}
                      style={{ fontSize: '12px', color: '#34d399', textDecoration: 'underline' }}
                    >
                      Download
                    </a>
                  )}
                </div>
                {activeTask.proof.data && activeTask.proof.data.startsWith('data:image') && (
                  <img
                    src={activeTask.proof.data}
                    alt="Proof Preview"
                    style={{ marginTop: '10px', maxWidth: '100%', maxHeight: '200px', borderRadius: '6px', border: '1px solid #059669' }}
                  />
                )}
              </div>
            )}

            <div>
              <h4 style={{ margin: '0 0 10px', fontSize: '12px', textTransform: 'uppercase', letterSpacing: '0.05em', color: '#94a3b8' }}>Activity & Comments</h4>
              <div style={{ display: 'flex', flexDirection: 'column', gap: '8px', maxHeight: '200px', overflowY: 'auto', marginBottom: '14px' }}>
                {activeTask.comments.map(c => (
                  <div key={c.id} style={{ background: c.isSignOff ? '#1e3a5f' : '#1e293b', border: c.isSignOff ? '1px solid #3b82f6' : '1px solid #334155', borderRadius: '8px', padding: '10px 12px' }}>
                    <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: '11px', color: '#64748b', marginBottom: '4px' }}>
                      <span style={{ fontWeight: 600, color: c.isSignOff ? '#60a5fa' : '#94a3b8' }}>{c.author} {c.isSignOff && '• Verified Sign-off'}</span>
                      <span>{c.time}</span>
                    </div>
                    <div style={{ fontSize: '13px', color: '#e2e8f0', lineHeight: 1.4 }}>{c.text}</div>
                    {c.proofAttachment && c.proofAttachment.data && c.proofAttachment.data.startsWith('data:image') && (
                      <img src={c.proofAttachment.data} alt="Proof" style={{ marginTop: '8px', maxHeight: '120px', borderRadius: '4px' }} />
                    )}
                  </div>
                ))}
                {activeTask.comments.length === 0 && (
                  <div style={{ color: '#475569', fontSize: '12px', fontStyle: 'italic' }}>No comments recorded yet.</div>
                )}
              </div>

              <div style={{ background: '#1e293b', border: '1px solid #334155', borderRadius: '8px', padding: '12px', display: 'flex', flexDirection: 'column', gap: '10px' }}>
                <textarea
                  value={commentInput}
                  onChange={(e) => setCommentInput(e.target.value)}
                  placeholder="Leave an update, status report, or notes..."
                  style={{ background: '#0f172a', border: '1px solid #334155', borderRadius: '6px', color: '#f8fafc', padding: '8px 10px', fontSize: '13px', minHeight: '60px', resize: 'vertical' }}
                />

                <div style={{ display: 'flex', alignItems: 'center', justifyContent: 'space-between', flexWrap: 'wrap', gap: '10px' }}>
                  <label style={{ display: 'inline-flex', alignItems: 'center', gap: '6px', background: '#0f172a', border: '1px solid #334155', padding: '6px 12px', borderRadius: '6px', fontSize: '12px', cursor: 'pointer', color: '#cbd5e1' }}>
                    📎 {proofFileName ? proofFileName : 'Attach proof (screenshot / PDF)'}
                    <input type="file" onChange={handleFileUpload} style={{ display: 'none' }} />
                  </label>

                  <div style={{ display: 'flex', gap: '8px' }}>
                    <button
                      onClick={() => submitComment(false)}
                      style={{ background: '#334155', color: '#f8fafc', border: 0, padding: '7px 14px', borderRadius: '6px', fontSize: '12px', fontWeight: 600, cursor: 'pointer' }}
                    >
                      Comment
                    </button>
                    <button
                      onClick={() => submitComment(true)}
                      style={{ background: '#16a34a', color: '#fff', border: 0, padding: '7px 14px', borderRadius: '6px', fontSize: '12px', fontWeight: 600, cursor: 'pointer' }}
                    >
                      ✓ Task is complete (with proof)
                    </button>
                  </div>
                </div>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* Create Task Modal */}
      {showNewTaskModal && (
        <div style={{ position: 'fixed', inset: 0, background: 'rgba(0,0,0,0.7)', display: 'flex', justifyContent: 'center', alignItems: 'center', zIndex: 50, padding: '20px' }}>
          <div style={{ background: '#0f172a', border: '1px solid #334155', borderRadius: '12px', width: '100%', maxWidth: '500px', padding: '24px', display: 'flex', flexDirection: 'column', gap: '14px' }}>
            <h3 style={{ margin: 0, fontSize: '16px', color: '#f8fafc' }}>Create New Task in {activeProject.name}</h3>
            
            <div>
              <label style={{ fontSize: '12px', color: '#94a3b8', display: 'block', marginBottom: '4px' }}>Title *</label>
              <input
                type="text"
                value={newTaskTitle}
                onChange={(e) => setNewTaskTitle(e.target.value)}
                placeholder="e.g. Implement AWS Bedrock ingestion Lambda"
                style={{ width: '100%', boxSizing: 'border-box', background: '#1e293b', border: '1px solid #334155', borderRadius: '6px', color: '#f8fafc', padding: '8px 10px', fontSize: '13px' }}
              />
            </div>

            <div>
              <label style={{ fontSize: '12px', color: '#94a3b8', display: 'block', marginBottom: '4px' }}>Description</label>
              <textarea
                value={newTaskDesc}
                onChange={(e) => setNewTaskDesc(e.target.value)}
                placeholder="Details, acceptance criteria, or requirements..."
                style={{ width: '100%', boxSizing: 'border-box', background: '#1e293b', border: '1px solid #334155', borderRadius: '6px', color: '#f8fafc', padding: '8px 10px', fontSize: '13px', minHeight: '80px' }}
              />
            </div>

            <div>
              <label style={{ fontSize: '12px', color: '#94a3b8', display: 'block', marginBottom: '4px' }}>Priority</label>
              <select
                value={newTaskPriority}
                onChange={(e) => setNewTaskPriority(e.target.value)}
                style={{ width: '100%', background: '#1e293b', border: '1px solid #334155', borderRadius: '6px', color: '#f8fafc', padding: '8px 10px', fontSize: '13px' }}
              >
                <option value="High">High</option>
                <option value="Medium">Medium</option>
                <option value="Low">Low</option>
              </select>
            </div>

            <div style={{ display: 'flex', justifyContent: 'flex-end', gap: '10px', marginTop: '10px' }}>
              <button
                onClick={() => setShowNewTaskModal(false)}
                style={{ background: 'transparent', border: '1px solid #334155', color: '#94a3b8', padding: '7px 14px', borderRadius: '6px', fontSize: '12px', cursor: 'pointer' }}
              >
                Cancel
              </button>
              <button
                onClick={handleCreateTask}
                style={{ background: '#2563eb', border: 0, color: '#fff', padding: '7px 14px', borderRadius: '6px', fontSize: '12px', fontWeight: 600, cursor: 'pointer' }}
              >
                Create Task
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
